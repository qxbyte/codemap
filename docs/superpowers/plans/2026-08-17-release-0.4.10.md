# CodeMap 0.4.10 Release Hardening Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Publish an installable and preflight-validated 21-package CodeMap release to PyPI.

**Architecture:** Distribution metadata remains in each `pyproject.toml`; each plugin has a local `_version.py` reader so it remains compatible with its declared minimum core version, while release-contract tests compare all package metadata with the GitHub publish matrix. CI adds an isolated package matrix so every plugin test suite and every wheel/sdist is validated before a release tag.

**Execution correction:** Pre-push review proved that placing the version helper in
`codemap-core 0.4.10` would break plugins installed with their advertised
`codemap-core>=0.4.0` minimum. The implementation therefore keeps the tiny
`importlib.metadata` reader inside each plugin and verifies a built `0.4.10`
plugin against the published `codemap-core 0.4.6`.

**Tech Stack:** Python 3.11+, Hatchling, pytest, packaging, Ruff, mypy, import-linter, GitHub Actions, build, twine, PyPI Trusted Publishing.

## Global Constraints

- Publish `codemap-core` and the 19 lockstep plugins as `0.4.10`.
- Publish `codemap-semantic-index` as `0.2.3`.
- Set lockstep plugin compatibility to `codemap-core>=0.4.0,<0.5`.
- Keep semantic index compatibility at `codemap-core>=0.4.1` and `codemap-aimemory>=0.4.1`.
- Do not change parser, recall, persistence, or CLI behavior beyond accurate reported versions.
- Every commit must contain Codex and repository-user co-author trailers.

```python
from __future__ import annotations

import tomllib
from pathlib import Path

import yaml
from packaging.requirements import Requirement
from packaging.utils import canonicalize_name
from packaging.version import Version

ROOT = Path(__file__).parents[2]
CORE_VERSION = "0.4.10"
SEMANTIC_VERSION = "0.2.3"


def _metadata(path: Path) -> dict[str, object]:
    with path.open("rb") as stream:
        return tomllib.load(stream)["project"]


def _package_projects() -> dict[str, tuple[Path, dict[str, object]]]:
    paths = [ROOT / "pyproject.toml", *sorted((ROOT / "plugins").glob("*/pyproject.toml"))]
    return {str(_metadata(path)["name"]): (path, _metadata(path)) for path in paths}


def test_publish_matrix_matches_repository_packages() -> None:
    workflow = yaml.safe_load((ROOT / ".github/workflows/publish.yml").read_text())
    rows = workflow["jobs"]["publish"]["strategy"]["matrix"]["package"]
    actual = {(row["name"], row["dir"]) for row in rows}
    expected = {
        (name, "." if path == ROOT / "pyproject.toml" else str(path.parent.relative_to(ROOT)))
        for name, (path, _project) in _package_projects().items()
    }
    assert actual == expected


def test_release_versions_and_core_compatibility() -> None:
    projects = _package_projects()
    assert projects["codemap-core"][1]["version"] == CORE_VERSION
    for name, (_path, project) in projects.items():
        expected = SEMANTIC_VERSION if name == "codemap-semantic-index" else CORE_VERSION
        assert project["version"] == expected
        if name == "codemap-core":
            continue
        core_req = next(
            Requirement(raw)
            for raw in project.get("dependencies", [])
            if canonicalize_name(Requirement(raw).name) == "codemap-core"
        )
        assert Version(CORE_VERSION) in core_req.specifier


def test_plugin_versions_are_metadata_driven() -> None:
    for name, (path, _project) in _package_projects().items():
        if name == "codemap-core":
            continue
        module_dir = next((path.parent / "src").iterdir())
        init_text = (module_dir / "__init__.py").read_text()
        assert f'package_version("{name}")' in init_text
        for source in module_dir.rglob("*.py"):
            text = source.read_text()
            if "version: ClassVar[str]" in text:
                assert "version: ClassVar[str] = __version__" in text
```

- [ ] **Step 1: Implement the minimal helper**

```python
"""Helpers for plugin distribution metadata."""

from __future__ import annotations

from importlib import metadata


def package_version(distribution: str) -> str:
    """Return an installed distribution version or the source-tree fallback."""
    try:
        return metadata.version(distribution)
    except metadata.PackageNotFoundError:
        return "0.0.0.dev0"
```
